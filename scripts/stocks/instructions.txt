# scripts/stocks — CLI tools (run from this folder)

# All scripts use JSON under this folder by default. No MONITOR_STORAGE required.

# Optional: STOCKS_USE_AZURE_STORAGE=1 (or quant-monitor --use-azure) to use Azure blob instead.



## momentum-analyzer.py — momentum portfolio (RS + trailing stop)

python momentum-analyzer.py --seed-from-finviz

python momentum-analyzer.py --seed-from-finviz --merge

python momentum-analyzer.py --tickers-file my_tickers.txt

python momentum-analyzer.py --tickers-file picks.txt --merge

python momentum-analyzer.py



Files: momentum-analyzer.json



## holdings-analyzer.py — holdings list trailing + RS (local JSON)

# RS ranks 6mo return; set peers-only (no SPY in rank) for “best grower in this list”:

python holdings-analyzer.py --set-from-file my_tickers.txt

python holdings-analyzer.py --set-from-file picks.txt --merge

python holdings-analyzer.py --set-from-file my_tickers.txt --push-azure

python holdings-analyzer.py --push-azure

python holdings-analyzer.py

python holdings-analyzer.py --remove-on-exit

cd scripts\stocks
python holdings-analyzer.py --set-from-file portfolio.csv --merge
python holdings-analyzer.py --set-from-file portfolio.csv --push-azure   # if using Azure

## quant-monitor.py — daily quant scores (full report to terminal; no CSV by default)

python quant-monitor.py

python quant-monitor.py --tickers AAPL MSFT

python quant-monitor.py --brief
python quant-monitor.py --write-csv

python quant-monitor.py --holdings-only

Merges (default): local-list.json, holdings-list.json, momentum-analyzer.json, Finviz sources (if enabled).

## pie_analyze_swing.py — swing + put-credit-spread funnel

# Scanner lives in pie_analyzer.py (run_scan / select_buy_candidates).
# This funnels BUY candidates into swing entry plan + PCS plan.

python pie_analyze_swing.py
python pie_analyze_swing.py --min-grade A --risk-per-trade 0.01 --target-dte 35 --otm-pct 0.06
python pie_analyze_swing.py --no-pcs

# PCS strikes by %-OTM from live yfinance option chain (no Greeks); POP ~ credit/width proxy.

# Lifecycle review of held positions (positions.json: swings + spreads):
python pie_analyze_swing.py --review
python pie_analyze_swing.py --review --positions positions.json

# positions.json shape:
#   {"swings": [{"symbol","entry_price","entry_date","shares","stop_price"}],
#    "spreads": [{"symbol","short_put","long_put","expiration","credit","entry_date"}]}
